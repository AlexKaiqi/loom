11. 接口类型与编码

本页是工作契约中的目标接口，定义事件、输入、模型、Harness 调用和展示快照的数据类型，供 Harness 作者及 SDK 实现者核对；不指定执行引擎或传输组件。TypeScript 声明描述输入、输出与确认边界；Promise 表达异步返回，持久性由所选执行绑定提供。

request_key 标识可跨连接核对的业务操作。Work 身份和执行所有权由受信入口确认，秘密不进入业务载荷。字节数、序号等大整数使用十进制字符串；大载荷通过摘要、大小及媒体类型组成的引用传递。进程内 SDK、CLI 与跨进程传输须保持同一语义，具体封装见SDK 与传输接合。

作者通过 event_type + data 表达事件内容，事件记录保留确定的 event_ref。来源、目标和消费位置由 Runtime 绑定，不能由正文覆盖；具体传输信封不改变这些确认边界。

inputs.transfer 的 processing_ref 引用已领取输入的处理身份，target_run_id 指定同一 Work 内的目标调用；返回的新处理身份绑定目标 scope，回执的持有者不能据此冒充目标处置。交接、领取、重复请求和关闭的完整规则见输入处理责任转交契约。

Event／Input 类型 · event-contracts.ts
// 目标接口；Promise 表达异步返回，运行绑定负责保存状态与恢复。
export type Json = null | boolean | number | string | Json[] | { [key: string]: Json };
export type Ref = { sha256: string; size_bytes: string; media_type: string };
export type EventRecord = {
  id: string;
  source: string;             // 已验证的来源逻辑身份
  event_type: string;
  position: string;           // 不透明来源游标，不按字符串比较
  data: Json;
  event_ref: Ref;
};
export type PublicationReceipt = {
  event_id: string;
  event_ref: Ref;
  position: string;
  status: 'published';
};
export type SubscribeRequest = {
  request_key: string;
  source: string;
  types: string[];            // 非空、无重复，精确匹配
  from: 'now' | { after: string };
};
export type SubscriptionReceipt = {
  subscription_id: string;
  start_cursor: string;
  status: 'active';
};
export type InputDelivery = {
  input: {
    id: string;
    ordinal: string;
    event: EventRecord;
    matched_subscriptions: string[];
  };
  processing_ref: string;     // Runtime 绑定；作者原样传回，不自行构造
};
export type NextInput =
  | { kind: 'input'; delivery: InputDelivery }
  | { kind: 'closed'; reason: 'subscription_closed' | 'work_closed' };
export type Disposition =
  | { outcome: 'handled'; result_ref?: Ref }
  | { outcome: 'failed' | 'ignored'; reason: string; result_ref?: Ref };
export interface EventPort {
  publish(request: {
    request_key: string; event_type: string; data: Json;
  }): Promise<PublicationReceipt>;
  subscribe(request: SubscribeRequest): Promise<SubscriptionReceipt>;
  unsubscribe(request: {
    request_key: string; subscription_id: string;
  }): Promise<{ subscription_id: string; status: 'closed'; receipt_ref: Ref }>;
  describe(request: { source: string }): Promise<Array<{
    event_type: string; schema_ref: Ref; meaning: string;
  }>>;
  inspect(subscriptionId: string): Promise<{
    definition: SubscribeRequest; start_cursor: string;
    state: 'preparing' | 'active' | 'closed' | 'blocked';
    observation_ref: Ref;
  }>;
  list(request: { cursor?: string; limit: number }): Promise<{
    subscription_ids: string[]; next_cursor?: string;
  }>;
}
export interface InputPort {
  // 运行绑定保存稳定 next 身份；候选仅含本 scope 获准的邮箱输入或明确转交的输入。
  next(filter: { subscription_id?: string; types?: string[] }): Promise<NextInput>;
  // 同时核对当前 run scope、有效 processing_ref 和执行所有权；旧身份不能提交新结果。
  complete(delivery: InputDelivery, result: Disposition): Promise<{
    input_id: string; outcome: Disposition['outcome']; receipt_ref: Ref;
  }>;
  // 原处理责任在同一 Work 内交给目标调用；目标凭受信 run scope 领取新 delivery。
  transfer(request: {
    request_key: string; processing_ref: string; target_run_id: string;
  }): Promise<{
    input_id: string; target_run_id: string;
    processing_ref: string; // 新身份绑定目标 scope；来源拿到回执也不能冒充目标处置
    receipt_ref: Ref;
  }>;
}
export type ContractFailure = {
  kind: 'invalid_request' | 'permission_denied' | 'identity_conflict'
    | 'history_gap' | 'work_closed' | 'subscription_closed'
    | 'stale_processing' | 'not_ready' | 'resource_exhausted'
    | 'outcome_unknown' | 'internal_error'
    | 'invalid_code' | 'dependency_unavailable' | 'unsupported_binding'
    | 'recovery_incompatible' | 'call_limit_exceeded';
  message: string;
  request_key?: string;
  evidence_ref?: Ref;
};
Model 类型 · model-contracts.ts
import type { Json, Ref } from './event-contracts';
// 模型目标契约；不是已发布 SDK。
// Promise 类型仅表达异步接口；持久性须由执行绑定实现并验证。
export type Message = {
  role: 'system' | 'developer' | 'user' | 'assistant' | 'tool';
  content: Json; // 媒体、工具关系与原生块按锁定的消息 schema 验证
};
export type ModelInput =
  | { messages: Message[]; messages_ref?: never; projection_ref?: never }
  | { messages_ref: Ref; messages?: never; projection_ref?: never }
  | { projection_ref: Ref; messages?: never; messages_ref?: never };
export type OutputContract =
  | { mode: 'text' }
  | { mode: 'json_schema'; schema_ref: Ref;
      enforcement: 'provider_required' | 'validate_result' };
export type ModelRequest = {
  schema_version: 1;
  request_key: string;
  model: { service: string; model_id?: string; parameters?: Record<string, Json> };
  input: ModelInput;
  output: OutputContract;
  tools_ref?: Ref;
  limits: { max_output_tokens: number; budget_ref: string };
  cache?: { mode: 'provider_default' | 'prefer_prefix'; prefix_ref?: Ref };
  attribution?: { stage?: string; item?: string; pass?: number };
};
export type ModelCallOutcome = {
  operation_id: string;
  selected_attempt_id: string;
  request_ref: Ref;
  response_ref: Ref;
  stop_reason: 'stop' | 'tool_use' | 'length' | 'error' | 'aborted';
  tool_calls: Array<{ id: string; name: string; arguments: Json }>;
  validation: 'not_requested' | 'valid' | 'invalid' | 'unavailable';
  output_ref?: Ref;
  usage_ref: Ref; // 供应商未报告的值标 unknown，不填零
};
export type OperationReceipt = {
  operation_id: string;
  request_ref: Ref;
  binding_ref: Ref;
  status: 'accepted';
};
export interface ModelPort {
  // 成功仅承诺原请求和执行责任持久成立；相同 key 不创建第二次操作。
  submit(request: ModelRequest): Promise<OperationReceipt>;
  // 在持久执行绑定中等待已选定结果；不执行返回的工具，不启动新推理。
  result(operationId: string): Promise<ModelCallOutcome>;
  inspect(operationId: string): Promise<{
    status: 'pending' | 'running' | 'unknown' | 'complete';
    cancellation: 'none' | 'requested' | 'stop_confirmed' | 'unknown';
    observation_ref: Ref; result?: ModelCallOutcome;
  }>;
  cancel(arg: { operation_id: string; request_key: string }): Promise<{
    operation_id: string; status: 'cancellation_requested'; receipt_ref: Ref;
  }>;
  describe(model: ModelRequest['model']): Promise<{
    capability_ref: Ref; observed_at: string; // 身份、窗口与支持范围在记录中
  }>;
  // submit + result 的便利入口，不包含 agent loop 或隐式重试。
  invoke(request: ModelRequest): Promise<ModelCallOutcome>;
}

Harness 调用

code_ref 指向第 8 章的不可变代码描述；entry 必须是 entries 许可入口，SDK、依赖和必需能力由 Runtime 与宿主 Profile 核对，不能据此自授权。状态推进入口必须声明状态 schema；顺序入口的恢复格式由其执行绑定固定。start 保存原请求、解析后的固定绑定和调用责任后返回受理回执；同键改代码、输入、环境或限制返回 identity_conflict。Work、父调用、执行所有权和授权来源均由受信 scope 确认，正文不能伪造。result 等待已保存结果；业务终态与未结操作责任分别表达，未知外部效果不能被 cancelled 覆盖。tree 仅指同一 Work 的调用后代，独立 Work 的取消遵循其授权。

Harness 类型 · harness-contracts.ts
import type { Ref } from './event-contracts';
import type { OperationReceipt } from './model-contracts';
// 外部 CLI／RPC 必须传 request_key；受管 SDK 从调用日志分配同一逻辑调用的键。
export type HarnessLimits = {
  budget_ref?: string;
  targets?: string[];
  model_services?: string[];
  max_parallel_calls?: number; // 正整数，省略即沿用已授权上限
  max_depth?: number;          // 正整数，省略即沿用已授权上限
};
export type HarnessStartRequest = {
  request_key: string;
  code_ref: Ref;
  entry: string;
  input_ref: Ref;
  environment: string;        // Work 获准的逻辑 Harness 环境
  limits?: HarnessLimits;     // 只能收窄 Work／父调用的实际范围
};
export type HarnessBinding = {
  run_id: string;
  code_ref: Ref;
  entry: string;
  input_ref: Ref;
  runtime_profile: Ref;       // 已解析运行时、Linux 隔离与资源要求快照
  sdk_ref: Ref;
  dependency_ref: Ref;
  binding_ref: Ref;           // 固定执行引擎、适配器与恢复入口
  compiled_ref?: Ref;         // TS／编译场景固定实际装载产物
  effective_scope_ref: Ref;
} & (
  | { mode: 'managed_sequence'; state_schema?: number }
  | { mode: 'state_step'; state_schema: number }
);
export type HarnessStartReceipt = OperationReceipt & {
  run_id: string;
  receipt_ref: Ref;
};
export type HarnessResponsibility =
  | { status: 'settled'; unsettled_operation_ids: [] }
  | { status: 'transferred'; unsettled_operation_ids: string[];
      transfer_receipt_ref: Ref };
export type HarnessOutcome = {
  run_id: string;
  business_result: {
    outcome: 'succeeded' | 'failed' | 'cancelled';
    result_ref?: Ref;
    reason_ref?: Ref;
  };
  receipt_ref: Ref;
  responsibility: HarnessResponsibility;
};
export interface HarnessPort {
  start(request: HarnessStartRequest): Promise<HarnessStartReceipt>;
  result(request: { run_id: string }): Promise<HarnessOutcome>;
  inspect(request: { run_id: string }): Promise<{
    run_id: string;
    binding: HarnessBinding;
    status: 'accepted' | 'running' | 'waiting' | 'blocked' | 'unknown' | 'complete';
    cancellation: 'none' | 'requested' | 'stop_confirmed' | 'unknown';
    blocked_reason_ref?: Ref;
    unsettled_operation_ids: string[];
    observation_ref: Ref;
    result?: HarnessOutcome;
  }>;
  cancel(request: {
    request_key: string; run_id: string; scope: 'self' | 'tree';
  }): Promise<{
    operation_id: string; run_id: string; scope: 'self' | 'tree';
    status: 'cancellation_requested'; receipt_ref: Ref;
  }>;
}
export interface ManagedHarnessPort extends Omit<HarnessPort, 'start' | 'cancel'> {
  start(request: Omit<HarnessStartRequest, 'request_key'> & {
    request_key?: string;
  }): Promise<HarnessStartReceipt>;
  cancel(request: {
    request_key?: string; run_id: string; scope: 'self' | 'tree';
  }): ReturnType<HarnessPort['cancel']>;
}

所有操作复用共同 ContractFailure。代码或入口不能验证返回 invalid_code；依赖无法取得返回 dependency_unavailable；作者形式或隔离 SDK 接合不支持返回 unsupported_binding；原版本恢复约束不相容返回 recovery_incompatible；超过获准深度或并发返回 call_limit_exceeded。配置尚未解析是 not_ready，已获准资源耗尽是 resource_exhausted；已派发但无法核对的调用保持 outcome_unknown。受理前拒绝不得留下可独立执行的第二份调用。深度与并发按已保存的调用树和 Work 配额核对，子调用不能通过重新声明限制重置上限；tree 取消与新后代受理由同一控制顺序裁决。

调用深度、并发和名额释放统一按Harness 调用限额定义；DTO 只表达请求的收窄值，不另定义计数规则。

显式状态作者入口

完整 ABI 与类型定义见第 8 章显式状态入口；受管顺序入口见同章受管顺序入口。本页的调用与结果类型供这两种作者形式共用。

展示快照

来源范围、图中判断的依据及更新竞争按图与结构展示契约处理。graph 不另复制图组件的节点、边和结构类型；载入前调用其公开校验入口。快照先保存为 Content 引用，再通过既有事件发布该引用与范围，不添加另一套图调度或存储权威。

Graph 类型 · graph-contracts.ts
import type { Json, Ref } from './event-contracts';
// Loom 只定义来源信封；内部 graph/view 按所固定图组件契约校验。
export type GraphPublication = {
  format: 'loom.graph';
  version: 1;
  source: {
    work_id: string;          // 受信发布范围核对，不能据此取得权限
    producer_ref: Ref;
    view_id: string;          // 同 producer 可提供多个命名视图
    definition_ref: Ref;
    revision: number;         // 正安全整数，由持久来源记录分配
    records: {
      after?: string;         // 排他；与 through 同属记录来源
      through: string;        // 包含；游标不做词法比较
      coverage: 'complete' | 'partial';
    };
  };
  snapshot: {
    format: 'graph-workbench';
    version: 1;
    graph: Json;
    view: { collapsed: string[] };
  };
};
